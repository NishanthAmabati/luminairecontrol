API Endpoints:

http://localhost:8000/docs#/

GET /status: Get system status (CCT, intensity, auto_mode, connected devices, etc.).
GET /status/luminaires: Get connected luminaires.
POST /set_cct: Set color temperature (CCT):
POST /set_intensity: Set intensity.
POST /toggle_system: Turn system on/off.
POST /set_mode: Switch between auto/manual mode.
POST /activate_scene: Activate a scene.
POST /load_scene: Load a scene for preview.
POST /stop_scheduler: Stop the running scheduler.

curl -X GET 'http://localhost:8000/status'
curl -X GET 'http://localhost:8000/status/luminaires'
curl -X GET 'http://localhost:8000/status/cct' 
curl -X GET 'http://localhost:8000/status/intensity'
 
curl -X POST 'http://localhost:8000/set_cct' -H 'Content-Type: application/json' -d '{"cct": 4000}'
curl -X POST 'http://localhost:8000/set_intensity' -H 'Content-Type: application/json' -d '{"intensity":500}'
curl -X POST 'http://localhost:8000/set_mode' -H 'Content-Type: application/json' -d '{"auto": true}'
curl -X POST 'http://localhost:8000/toggle_system' -H 'Content-Type: application/json' -d '{"isSystemOn": true}'
curl -X POST 'http://localhost:8000/load_scene' -H 'Content-Type: application/json' -d '{"scene": "scene5.csv"}'
curl -X POST 'http://localhost:8000/activate_scene' -H 'Content-Type: application/json' -d '{"scene": "scene5.csv"}'
curl -X POST 'http://localhost:8000/stop_scheduler'
